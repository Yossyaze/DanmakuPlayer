    event.target.value = '';
  };

  const handleRemoveFile = (fileId) => {
    setLoadedFiles(prev => prev.filter(f => f.id !== fileId));
  };

  useEffect(() => {
    if (loadedFiles.length === 0) {
      setComments([]);
      return;
    }
    let allRawComments = loadedFiles.flatMap(f => f.rawComments);
    allRawComments.sort((a, b) => a.rawTime - b.rawTime);

    const calculateVideoTimes = (rawComments, startStr, fallbackBaseTime) => {
      let baseTimeMs = fallbackBaseTime;
      if (startStr && rawComments.length > 0) {
        const firstLogDate = new Date(rawComments[0].rawTime);
        const [h, m, s] = startStr.split(':').map(Number);
        if (!isNaN(h) && !isNaN(m) && !isNaN(s)) {
          const userBaseDate = new Date(firstLogDate);
          userBaseDate.setHours(h, m, s, 0);
          baseTimeMs = userBaseDate.getTime();
        }
      }
      return rawComments.map(c => ({ ...c, time: (c.rawTime - baseTimeMs) / 1000 }));
    };
    const calculated = calculateVideoTimes(allRawComments, startTimeStr, allRawComments[0]?.rawTime);
    setComments(calculated);
  }, [loadedFiles, startTimeStr]);

  const currentLogicalTime = useMemo(() => {
    return currentTime + totalWaitOffset + currentCmWaitTime;
  }, [currentTime, totalWaitOffset, currentCmWaitTime]);

  const activeCommentId = useMemo(() => {
    let activeId = null;
    for (let i = 0; i < comments.length; i++) {
      if (comments[i].time > currentLogicalTime) break;
      activeId = comments[i].id;
    }
    return activeId;
  }, [comments, currentLogicalTime]);

  const currentThreadTitle = useMemo(() => {
    if (loadedFiles.length === 0) return "";
    let activeFileId = null;
    if (activeCommentId) {
      const activeComment = comments.find(c => c.id === activeCommentId);
      if (activeComment) activeFileId = activeComment.sourceFileId;
    } 
    if (!activeFileId && comments.length > 0) activeFileId = comments[0].sourceFileId;

    const file = loadedFiles.find(f => f.id === activeFileId);
    const titleStr = file ? (file.title || file.name) : (loadedFiles[0].title || loadedFiles[0].name);
    return String(titleStr); 
  }, [activeCommentId, comments, loadedFiles]);

  const scrollToActive = useCallback(() => {
    if (activeCommentRef.current && scrollContainerRef.current) {
      const container = scrollContainerRef.current;
      const element = activeCommentRef.current;
      const elementBottom = element.offsetTop + element.offsetHeight;
      const containerHeight = container.clientHeight;
      const scrollTarget = elementBottom - containerHeight + 8;
      container.scrollTo({ top: scrollTarget, behavior: 'smooth' });
    }
  }, []);

  const handleUserScrollInteraction = () => {
    if (isAutoScroll) setIsAutoScroll(false);
    setShowScrollButton(true);
  };

  const handleSyncButton = () => {
    setIsAutoScroll(true);
    setShowScrollButton(false);
    scrollToActive();
  };

  useEffect(() => {
    if (isAutoScroll) {
      scrollToActive();
    }
  }, [currentLogicalTime, isAutoScroll, scrollToActive, activeCommentId]); 

  useEffect(() => {
    const cleanupTime = dmSettings.duration + 2;
    const interval = setInterval(() => {
      setActiveDanmaku(prev => prev.filter(d => d.time > currentLogicalTime - cleanupTime));
    }, 1000);
    return () => clearInterval(interval);
  }, [currentLogicalTime, dmSettings.duration]);

  const formatTime = (time) => {
    const absTime = Math.abs(time);
    const sign = time < 0 ? "-" : "";
    const minutes = Math.floor(absTime / 60);
    const seconds = Math.floor(absTime % 60);
    return `${sign}${minutes}:${seconds.toString().padStart(2, '0')}`;
  };
  const getParentComment = (text) => {
    if (!enableTreeView) return null;
    const match = text.match(ANCHOR_REGEX);
    if (match) {
      const targetResNum = parseInt(match[1]);
      return comments.find(c => c.originalResNum === targetResNum);
    }
    return null;
  };
  const renderCommentContent = (text) => {
    const parts = text.split(URL_REGEX);
    return parts.map((part, index) => {
      if (part.match(/^https?:\/\//)) {
        const isImage = part.match(IMG_EXT_REGEX);
        return (
          <React.Fragment key={index}>
            <a href={part} target="_blank" rel="noopener noreferrer" className="text-blue-400 hover:underline inline-flex items-center gap-1 break-all">
              {part} <ExternalLink size={10} />
            </a>
            {showImages && isImage && (
              <div className="mt-2 mb-2">
                <img src={part} alt="image" className="max-w-full h-auto max-h-64 rounded border border-gray-700 bg-black/50 block" referrerPolicy="no-referrer" loading="lazy" onError={(e) => { e.target.style.display = 'none'; }} />
              </div>
            )}
          </React.Fragment>
        );
      }
      return <span key={index}>{part}</span>;
    });
  };

  return (
    <div className={`flex flex-col h-screen bg-gray-900 text-white font-sans overflow-hidden ${isResizing ? 'cursor-col-resize select-none' : ''}`} onMouseMove={resize} onMouseUp={stopResizing}>
      {/* Header */}
      <header className="bg-gray-800 p-3 flex items-center justify-between shadow-md z-10 min-h-[3.5rem] shrink-0">
        <div className="flex items-center gap-2 flex-1 min-w-0 mr-4">
          <Monitor className="text-blue-400 shrink-0" />
          <h1 className="text-lg font-bold hidden sm:block shrink-0">Danmaku Player</h1>
          {showThreadTitle && currentThreadTitle && (
            <div className="ml-2 md:ml-4 px-3 py-1 bg-gray-700 rounded text-xs text-gray-200 border-l-2 border-blue-500 whitespace-pre-wrap break-all leading-tight max-h-20 overflow-y-auto scrollbar-thin">
              {typeof currentThreadTitle === 'string' ? currentThreadTitle : ''}
            </div>
          )}
        </div>
        <div className="flex gap-2 text-xs sm:text-sm shrink-0">
           <button onClick={() => setMode("overlay")} className={`px-3 py-1 rounded ${mode === 'overlay' ? 'bg-blue-600' : 'bg-gray-700'}`}>弾幕のみ</button>
           <button onClick={() => setMode("sidebar")} className={`px-3 py-1 rounded ${mode === 'sidebar' ? 'bg-blue-600' : 'bg-gray-700'}`}>ログのみ</button>
           <button onClick={() => setMode("both")} className={`px-3 py-1 rounded ${mode === 'both' ? 'bg-blue-600' : 'bg-gray-700'}`}>両方</button>
        </div>
      </header>

      {/* Main Content */}
      <div className="flex flex-1 overflow-hidden relative">
        <div className={`flex flex-col relative bg-black justify-center transition-none`} style={{ width: (mode === 'sidebar' || mode === 'both') ? `calc(100% - ${sidebarWidth}px)` : '100%', pointerEvents: isResizing ? 'none' : 'auto' }}>
          {!videoSrc ? (
            <div className="flex flex-col items-center justify-center h-full text-gray-400 gap-6 p-8">
              <div className="bg-gray-800 p-8 rounded-xl border-2 border-dashed border-gray-600 w-full max-w-md text-center">
                <h2 className="text-xl font-bold mb-4 text-white">動画を読み込む</h2>
                <div className="flex gap-4 justify-center mb-6">
                  <button onClick={() => setInputType('file')} className={`flex items-center gap-2 px-4 py-2 rounded ${inputType === 'file' ? 'bg-blue-600 text-white' : 'bg-gray-700'}`}><Upload size={18} /> ファイル</button>
                  <button onClick={() => setInputType('url')} className={`flex items-center gap-2 px-4 py-2 rounded ${inputType === 'url' ? 'bg-blue-600 text-white' : 'bg-gray-700'}`}><LinkIcon size={18} /> URL</button>
                </div>
                {inputType === 'file' ? (
                  <label className="cursor-pointer bg-gray-700 hover:bg-gray-600 transition px-6 py-3 rounded-lg inline-flex items-center gap-2">
                    <Upload size={20} /><span>動画を選択 (MP4)</span>
                    <input type="file" accept="video/*" onChange={handleFileChange} className="hidden" />
                  </label>
                ) : (
                  <form onSubmit={handleUrlSubmit} className="flex gap-2">
                    <input type="text" placeholder="https://example.com/video.mp4" className="flex-1 bg-gray-900 border border-gray-600 rounded px-3 py-2 text-white" value={urlInput} onChange={(e) => setUrlInput(e.target.value)} />
                    <button type="submit" className="bg-blue-600 px-4 py-2 rounded hover:bg-blue-500">Load</button>
                  </form>
                )}
              </div>
            </div>
          ) : (
            <div ref={containerRef} className="relative w-full h-full bg-black flex items-center justify-center overflow-hidden group">
              <video 
                ref={videoRef} 
                src={videoSrc} 
                className="max-h-full max-w-full" 
                onClick={togglePlay}
                onLoadedMetadata={handleLoadedMetadata} 
                onEnded={() => setIsPlaying(false)} 
              />
              {(mode === 'overlay' || mode === 'both') && (
                <div className="absolute inset-0 pointer-events-none overflow-hidden select-none">
                  {activeDanmaku.map((d) => (
                    <div key={d.tempId} className="absolute whitespace-nowrap font-bold animate-danmaku" style={{ top: d.top, color: d.color, fontSize: `${dmSettings.fontSize}px`, opacity: dmSettings.opacity, textShadow: `1px 1px 0 #000, -1px -1px 0 #000, 1px -1px 0 #000, -1px 1px 0 #000, 1px 1px 2px black`, '--duration': `${dmSettings.duration}s`, animationPlayState: (isPlaying || isWaitingCm) ? 'running' : 'paused', transform: 'translateZ(0)', backfaceVisibility: 'hidden' }}>
                      {d.text}
                    </div>
                  ))}
                </div>
              )}
              {/* CM Wait Indicator */}
              {isWaitingCm && (
                 <div className="absolute bottom-20 right-4 bg-yellow-600/90 text-white px-3 py-2 rounded flex items-center gap-2 shadow-lg animate-pulse z-40 pointer-events-none">
                   <Coffee size={18} />
                   <span className="font-bold text-sm">CM待機中... ログ進行のみ</span>
                 </div>
              )}
              <div className="absolute bottom-0 left-0 right-0 bg-gradient-to-t from-black/80 to-transparent p-4 opacity-0 group-hover:opacity-100 transition-opacity duration-300">
                {/* Custom Seek Bar with Color Segments */}
                <div className="relative w-full h-1 mb-4 group/seek hover:h-2 transition-all cursor-pointer touch-none">
                    {/* Background Track */}
                    <div className="absolute inset-0 bg-gray-600 rounded-lg overflow-hidden">
                        {/* CM Ranges (Wait mode only) */}
                        {cmBehavior === 'wait' && cmRanges.map((range, i) => {
                            const startDisplay = getStaticDisplayTime(range.start);
                            const duration = range.end - range.start;
                            const leftPct = (startDisplay / getTotalDuration) * 100;
                            const widthPct = (duration / getTotalDuration) * 100;
                            return (
                                <div 
                                    key={i} 
                                    className="absolute top-0 h-full bg-yellow-500/70"
                                    style={{ left: `${leftPct}%`, width: `${widthPct}%` }}
                                    title={`CM: ${formatCmTime(range.start)}~`}
                                />
                            );
                        })}
                        {/* Progress Bar */}
                        <div 
                            className="absolute top-0 left-0 h-full bg-blue-500 pointer-events-none" 
                            style={{ width: `${(currentTime / getTotalDuration) * 100}%` }}
                        />
                    </div>
                    {/* Thumb */}
                    <div 
                        className="absolute top-1/2 -translate-y-1/2 w-3 h-3 bg-white rounded-full shadow pointer-events-none transform -translate-x-1/2 scale-0 group-hover/seek:scale-100 transition-transform"
                        style={{ left: `${(currentTime / getTotalDuration) * 100}%` }}
                    />
                    {/* Interactive Input */}
                    <input 
                        type="range" 
                        min="0" 
                        max={getTotalDuration || 0} 
                        value={currentTime} 
                        onChange={handleSeek} 
                        className="absolute inset-0 w-full h-full opacity-0 cursor-pointer" 
                    />
                </div>

                <div className="flex items-center justify-between">
                  <div className="flex items-center gap-4">
                    <button onClick={togglePlay} className="hover:text-blue-400 transition">{isPlaying ? <Pause size={24} /> : <Play size={24} />}</button>
                    <div className="flex items-center gap-2 group/vol">
                      <button onClick={() => setIsMuted(!isMuted)}>{isMuted ? <VolumeX size={20} /> : <Volume2 size={20} />}</button>
                      <input type="range" min="0" max="1" step="0.1" value={isMuted ? 0 : volume} onChange={(e) => { const v = parseFloat(e.target.value); setVolume(v); if(videoRef.current) videoRef.current.volume = v; setIsMuted(v === 0); }} className="w-20 h-1 bg-gray-600 rounded accent-white" />
                    </div>
                    <span className="text-sm font-mono">{formatTime(currentTime)} / {formatTime(getTotalDuration)}</span>
                  </div>
                </div>
              </div>
            </div>
          )}
        </div>
        {/* Resizer */}
        {(mode === 'sidebar' || mode === 'both') && (
          <div className="w-1.5 bg-gray-900 border-l border-r border-gray-800 hover:bg-blue-500 cursor-col-resize flex items-center justify-center z-30 transition-colors" onMouseDown={startResizing}>
            <GripVertical size={12} className="text-gray-600" />
          </div>
        )}
        {/* Sidebar */}
        {(mode === 'sidebar' || mode === 'both') && (
          <div className="bg-gray-900 flex flex-col shadow-xl z-20 relative" style={{ width: `${sidebarWidth}px` }}>
            <div className="flex items-center justify-between p-3 bg-gray-800 cursor-pointer border-b border-gray-700 hover:bg-gray-750 shrink-0" onClick={() => setShowSettingsPanel(!showSettingsPanel)}>
              <div className="flex items-center gap-2 font-bold text-gray-300 text-sm"><Settings size={16} /><span>設定・ログ読み込み</span></div>
              {showSettingsPanel ? <ChevronUp size={16} /> : <ChevronDown size={16} />}
            </div>
            {showSettingsPanel && (
              <div className="bg-gray-800 border-b border-gray-700 overflow-y-auto max-h-[60vh] scrollbar-thin shrink-0">
                <div className="p-4 space-y-5">
                  {/* 1. File Load */}
                  <div className="space-y-2">
                    <h4 className="text-xs font-bold text-gray-400 uppercase tracking-wider">ログ読み込み</h4>
                    <label className="flex flex-col items-center justify-center w-full h-16 px-4 transition bg-gray-700/50 border-2 border-gray-600 border-dashed rounded-md appearance-none cursor-pointer hover:border-gray-500 hover:bg-gray-700">
                      <span className="flex items-center space-x-2 text-gray-400"><Upload size={16} /><span className="text-sm">ログファイルを選択 (追加)</span></span>
                      <input type="file" accept=".txt" multiple className="hidden" onChange={handleLogFileChange} />
                    </label>
                    {loadedFiles.length > 0 && (
                      <div className="mt-2 space-y-1">
                        {loadedFiles.map(file => (
                          <div key={file.id} className="flex items-center justify-between bg-gray-900 p-2 rounded border border-gray-700 text-xs">
                            <div className="flex items-center gap-2 overflow-hidden"><FileText size={12} className="text-blue-400 shrink-0" /><span className="truncate text-gray-300">{file.name}</span><span className="text-gray-500 shrink-0">({file.rawComments.length})</span></div>
                            <button onClick={() => handleRemoveFile(file.id)} className="text-gray-500 hover:text-red-400 p-1 transition" title="削除"><Trash2 size={14} /></button>
                          </div>
                        ))}
                      </div>
                    )}
                  </div>
                  {/* 2. Sync & Display */}
                  <div className="space-y-2">
                    <h4 className="text-xs font-bold text-gray-400 uppercase tracking-wider">表示設定</h4>
                    <div className="flex items-center gap-2 bg-gray-900 p-2 rounded border border-gray-600 mb-2">
                      <Clock size={14} className="text-blue-400" />
                      <input type="time" step="1" value={startTimeStr} onChange={e => setStartTimeStr(e.target.value)} className="bg-transparent text-white text-sm flex-1 font-mono text-center focus:outline-none" />
                      <span className="text-xs text-gray-500">= 00:00</span>
                    </div>
                    <div className="space-y-1">
                      <label className="flex items-center gap-2 text-sm text-gray-300 cursor-pointer"><input type="checkbox" checked={showThreadTitle} onChange={e => setShowThreadTitle(e.target.checked)} className="rounded bg-gray-700 border-gray-600" /> スレッドタイトルを表示</label>
                      <label className="flex items-center gap-2 text-sm text-gray-300 cursor-pointer"><input type="checkbox" checked={enableTreeView} onChange={e => setEnableTreeView(e.target.checked)} className="rounded bg-gray-700 border-gray-600" /> アンカーをツリー表示 (引用)</label>
                      <label className="flex items-center gap-2 text-sm text-gray-300 cursor-pointer"><input type="checkbox" checked={showImages} onChange={e => setShowImages(e.target.checked)} className="rounded bg-gray-700 border-gray-600" /> 画像URLをインライン表示</label>
                    </div>
                  </div>
                   {/* CM Settings */}
                   <div className="space-y-2">
                    <div className="flex items-center justify-between">
                      <h4 className="text-xs font-bold text-gray-400 uppercase tracking-wider">CM区間設定</h4>
                      <div className="flex bg-gray-700 rounded p-0.5 text-[10px]">
                        <button 
                          onClick={() => { setCmBehavior('hidden'); setTotalWaitOffset(0); setIsWaitingCm(false); }} 
                          className={`px-2 py-0.5 rounded ${cmBehavior === 'hidden' ? 'bg-blue-600 text-white' : 'text-gray-400'}`}
                        >弾幕OFF</button>
                        <button 
                          onClick={() => { setCmBehavior('wait'); }} 
                          className={`px-2 py-0.5 rounded ${cmBehavior === 'wait' ? 'bg-blue-600 text-white' : 'text-gray-400'}`}
                        >動画停止&ログ進行</button>
                      </div>
                    </div>
                    <div className="flex items-center justify-between text-[10px] text-gray-400 mb-1">
                      <span>指定方法:</span>
                      <div className="flex bg-gray-700 rounded p-0.5">
                        <button onClick={() => setCmInputMode('video')} className={`px-2 py-0.5 rounded ${cmInputMode === 'video' ? 'bg-blue-500 text-white' : 'text-gray-400'}`}>動画時間</button>
                        <button onClick={() => setCmInputMode('log')} className={`px-2 py-0.5 rounded ${cmInputMode === 'log' ? 'bg-blue-500 text-white' : 'text-gray-400'}`}>ログ時刻</button>
                      </div>
                    </div>
                    <div className="flex items-center gap-2">
                      <input 
                        type="time" 
                        step="1" 
                        value={cmStartInput} 
                        onChange={e => setCmStartInput(e.target.value)} 
                        className="bg-gray-700 text-white text-xs p-2 rounded border border-gray-600 w-full font-mono text-center" 
                        placeholder="開始"
                      />
                      <span className="text-gray-400">~</span>
                      <input 
                        type="time" 
                        step="1" 
                        value={cmEndInput} 
                        onChange={e => setCmEndInput(e.target.value)} 
                        className="bg-gray-700 text-white text-xs p-2 rounded border border-gray-600 w-full font-mono text-center" 
                        placeholder="終了"
                      />
                      <button onClick={addCmRange} className="bg-blue-600 hover:bg-blue-500 text-white p-2 rounded"><Plus size={14} /></button>
                    </div>
                    {cmRanges.length > 0 && (
                      <div className="space-y-1 mt-1">
                        {cmRanges.map((range, i) => (
                          <div key={i} className="flex items-center justify-between bg-gray-900 p-1.5 rounded text-xs text-gray-400">
                            <span>
                              {range.mode === 'log' 
                                ? `ログ時刻: ${range.labelStart} ~ ${range.labelEnd}` 
                                : `動画時間: ${formatCmTime(range.start)} ~ ${formatCmTime(range.end)}`
                              }
                            </span>
                            <button onClick={() => removeCmRange(i)} className="hover:text-red-400"><X size={12} /></button>
                          </div>
                        ))}
                      </div>
                    )}
                  </div>
                  {/* 3. Danmaku Settings */}
                  <div className="space-y-3">
                    <h4 className="text-xs font-bold text-gray-400 uppercase tracking-wider">弾幕カスタマイズ</h4>
                    <div className="grid grid-cols-2 gap-4">
                      <div className="space-y-1"><label className="text-xs text-gray-400 flex justify-between">濃さ <span>{Math.round(dmSettings.opacity * 100)}%</span></label><input type="range" min="0.1" max="1" step="0.1" value={dmSettings.opacity} onChange={e => setDmSettings({...dmSettings, opacity: parseFloat(e.target.value)})} className="w-full h-1 bg-gray-600 rounded accent-blue-500" /></div>
                      <div className="space-y-1"><label className="text-xs text-gray-400 flex justify-between">サイズ <span>{dmSettings.fontSize}px</span></label><input type="range" min="12" max="64" step="2" value={dmSettings.fontSize} onChange={e => setDmSettings({...dmSettings, fontSize: parseInt(e.target.value)})} className="w-full h-1 bg-gray-600 rounded accent-blue-500" /></div>
                      <div className="space-y-1"><label className="text-xs text-gray-400 flex justify-between">速さ <span>{dmSettings.duration}秒</span></label><input type="range" min="3" max="20" step="1" value={dmSettings.duration} onChange={e => setDmSettings({...dmSettings, duration: parseInt(e.target.value)})} className="w-full h-1 bg-gray-600 rounded accent-blue-500" /></div>
                      <div className="space-y-1"><label className="text-xs text-gray-400 flex justify-between">表示範囲 <span>上{dmSettings.area}%</span></label><input type="range" min="10" max="100" step="10" value={dmSettings.area} onChange={e => setDmSettings({...dmSettings, area: parseInt(e.target.value)})} className="w-full h-1 bg-gray-600 rounded accent-blue-500" /></div>
                    </div>
                  </div>
                  <button onClick={() => setShowSettingsPanel(false)} className="w-full py-2 text-xs text-gray-400 bg-gray-700 hover:bg-gray-600 rounded flex items-center justify-center gap-1"><ChevronUp size={12} /> 設定を閉じる</button>
                </div>
              </div>
            )}
            {/* Comment List (ALL comments) */}
            <div ref={scrollContainerRef} onMouseDown={handleUserScrollInteraction} onTouchStart={handleUserScrollInteraction} className="flex-1 overflow-y-auto p-2 space-y-2 scrollbar-thin scrollbar-thumb-gray-600 bg-gray-900/95 relative">
              {comments.length === 0 ? (
                <div className="flex flex-col items-center justify-center h-64 text-gray-500 text-sm px-6 text-center"><FileText size={32} className="mb-2 opacity-50" /><p>「設定・ログ読み込み」から<br/>テキストファイル(.txt)を<br/>読み込んでください</p></div>
              ) : (
                comments.map((comment) => { 
                  const isActive = comment.id === activeCommentId;
                  const parent = getParentComment(comment.text);
                  return (
                    <div key={comment.id} ref={isActive ? activeCommentRef : null} className={`text-sm border-b border-gray-800 pb-2 last:border-0 ${comment.time > currentLogicalTime ? "opacity-40" : ""} ${isActive ? "bg-gray-800/50 -mx-2 px-2 py-1" : ""}`}>
                      {parent && (
                        <div className="mb-1 ml-4 pl-2 border-l-2 border-gray-600 text-xs text-gray-400 bg-gray-800/50 rounded p-1">
                          <div className="flex items-center gap-2 mb-0.5"><MessageSquare size={10} /><span>&gt;&gt;{parent.originalResNum}</span></div>
                          <div className="truncate opacity-80">{parent.text.split('\n')[0]}</div>
                        </div>
                      )}
                      <div className="flex gap-2">
                        <div className="flex flex-col items-end shrink-0 w-12 pt-1"><span className="text-gray-500 font-mono text-[10px]">{formatTime(comment.time)}</span><span className="text-gray-600 font-mono text-[10px]">{comment.dateDisplay?.split('.')[0]}</span></div>
                        <div className="flex-1 min-w-0">
                          <div className="text-[10px] text-gray-400 flex gap-2 mb-0.5"><span className="truncate max-w-[80px]">{comment.originalResNum}: 名無し</span></div>
                          <div className="text-white break-words whitespace-pre-wrap text-sm leading-snug">{renderCommentContent(comment.text)}</div>
                        </div>
                      </div>
                    </div>
                  );
                })
              )}
            </div>
            {/* Sync Button */}
            {!isAutoScroll && comments.length > 0 && (
              <button onClick={handleSyncButton} className="absolute bottom-6 right-6 bg-blue-600 hover:bg-blue-500 text-white p-3 rounded-full shadow-lg transition-all hover:scale-110 animate-bounce-small z-30 opacity-90" title="現在のコメントへ追従"><ArrowDown size={20} /></button>
            )}
          </div>
        )}
      </div>
      <style>{`

### 現在の弾幕実装アルゴリズム
現在のコード（[App.jsx](cci:7://file:///Users/kasugaiakira/Desktop/DanmakuPlayer/src/App.jsx:0:0-0:0)）は、以下のような仕組みで動いています。

1.  **監視ループ**: `requestAnimationFrame` というブラウザの機能を使って、1秒間に約60回、「今の動画時間は何秒か？」をチェックしています。
2.  **全探索**: 毎回、読み込んだ**全てのコメント**の中から「今の時間に表示すべきコメント」を検索（filter）しています。
3.  **DOM生成**: 表示すべきコメントが見つかると、Reactの機能を使って新しい `<div>` 要素（HTMLの箱）を作成し、画面に追加します。
4.  **CSSアニメーション**: 追加された `<div>` は、CSSの機能（`transform`）を使って右から左へ移動させます。
5.  **Reactの再描画**: ここが一番のボトルネックですが、動画の時間を更新するたび（`setCurrentTime`）、**画面全体（Appコンポーネント）の再描画（Re-render）** が走っています。

**重い理由:**
*   **Reactの更新頻度**: 1秒間に60回、画面全体の書き換え処理が走っているため、ブラウザの処理が追いつかず「カクつき」が発生します。
*   **DOMの増減**: HTML要素を作ったり消したりする処理は、コンピュータにとって比較的重い処理です。弾幕が増えるとこれが大量に発生します。

---

### ニコニコ動画などの高速化手法
ニコニコ動画や、高性能な弾幕プレイヤーは、以下のような技術を使って軽量化しています。

1.  **HTML5 Canvas描画**:
    *   `<div>` 要素を大量に作るのではなく、**`<canvas>` という「1枚の絵」** の上に、プログラムで文字を描画します。
    *   これにより、DOM要素は「1つ」で済むため、ブラウザの負荷が劇的に下がります。
2.  **バイナリサーチ（二分探索）**:
    *   コメントデータを時間順に並べ替えておき、「今の時間のコメント」を探す際に、全件検索ではなく高速な検索アルゴリズムを使います。
3.  **オブジェクトプーリング**:
    *   使い終わったコメントのデータ構造を捨てずに「再利用」することで、メモリの確保・解放（ガベージコレクション）による停止を防ぎます。

### 提案
もしよろしければ、現在の「DOM + CSSアニメーション」方式から、**「Canvas描画」方式**に書き換えることで、劇的に軽量化できる可能性があります。
